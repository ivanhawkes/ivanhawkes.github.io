{{/* Featured image for a page: "featured.webp" in the page's own resources, falling back to the parent page's resources. Alt text and the caption under the image come from the page's resource config: resources: - src: 'featured.webp' params: title: 'Lemon Bars' # caption under the image caption: 'A photograph of...' # alt text A page without that metadata falls back to the page title for the alt text, so alt is never empty. Callers pass a dict: "page" the page to render, and "caption" true to ask for the caption under the image, which page layouts do and card partials do not. */}} {{- $page := .page -}} {{- $showCaption := .caption -}} {{- $prefix := "featured.webp" -}} {{- $image := false -}} {{- with $page -}} {{- with .Resources -}} {{- $image = .GetMatch $prefix -}} {{- end -}} {{- if not $image -}} {{- with .Parent -}} {{- with .Resources -}} {{- $image = .GetMatch $prefix -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $image -}} {{- $meta := .Params | default dict -}} {{- $ctx := dict "image" . "destination" .RelPermalink "title" (cond $showCaption $meta.title "") "alt" ($meta.caption | default $page.LinkTitle) -}} {{- partial "/images/image-variants.html" $ctx -}} {{- end -}}
