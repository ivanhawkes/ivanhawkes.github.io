{{/* The head of one Kanban card page. The page used to lead with a copy of the card itself, which repeated the title the page carries and framed facts that are read, not scanned: the card belongs on the board, not at the top of its own page. This partial sets the same facts as a page head instead - the title with the priority, the ID and the state of the card on its line, the description under it, and every other fact as a definition list in the prose measure: two columns of label and value from the small screen up, one column below it. A row is drawn for a fact the card carries and not for one it does not, so a card with no due date, no references and no author shows none of them. A dependency or a reference that names a card of this board is a link to it; one that names a card not written stays the bare ID. */}} {{- $page := . -}} {{- $root := site.GetPage "section" "kanban" -}} {{/* The card's identity is its file name: the ID, the priority in its first character, and the stage in the folder it sits in. A topic folder borrows the number of the stage it belongs to, as the board's lanes do. */}} {{- $id := $page.File.BaseFileName -}} {{- $isCard := findRE `^p[0-9]-[a-z]{2}-[a-z0-9-]+$` $id -}} {{- $priority := int (slicestr $id 1 2) -}} {{- $kanbanDir := path.Clean $root.File.Dir -}} {{- $dir := path.Clean $page.File.Dir -}} {{- $folder := path.Base $dir -}} {{- $stage := $folder -}} {{- if ne (path.Dir $dir) $kanbanDir -}} {{- $stage = printf "%s-%s" (slicestr (path.Base (path.Dir $dir)) 0 2) $folder -}} {{- end -}} {{/* The state: done once a completion date is set, overdue when a due date has passed with the card still open. The stage the card sits in says the rest, so an open card on time states nothing of its own. */}} {{- $today := time.Format "2006-01-02" time.Now -}} {{- $completed := $page.Params.completionDate -}} {{- $overdue := false -}} {{- with $page.Params.due -}} {{- $overdue = and (not $completed) (gt $today (time.Format "2006-01-02" .)) -}} {{- end -}} {{/* A chip is a fill, so each one names the ink it carries: the brand and the state colours fill, and a priority that is not the urgent one is an outline, in the card's own ink. The classes are one string per chip, so no two of them ever paint the same property. */}} {{- $chip := "rounded-full border px-2 py-0.5 text-xs font-semibold whitespace-nowrap" -}} {{- $ink := "border-ink text-ink" -}} {{- $state := dict -}} {{- if $completed -}} {{- $state = dict "label" "Done" "colour" "border-success bg-success text-brand-ink" -}} {{- else if $overdue -}} {{- $state = dict "label" "Overdue" "colour" "border-danger bg-danger text-brand-ink" -}} {{- end -}} {{/* Every card of the board by its ID, so the lists below can link one. */}} {{- $byId := dict -}} {{- range $card := $root.RegularPagesRecursive -}} {{- with $card.File.BaseFileName -}} {{- $byId = merge $byId (dict . $card.RelPermalink) -}} {{- end -}} {{- end -}} {{/* A row of the list: the label in the left column, the value in the right. */}} {{- $label := "text-xs font-semibold uppercase tracking-wide text-ink-muted" -}} {{- $value := "text-ink" -}}
