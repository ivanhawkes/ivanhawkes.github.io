{{/* The card of one Kanban page, as a stage page lays it out. The card is scanned before it is read, so the frame carries what the words cannot reach from the far end of a lane: the priority, in the weight and the colour of the border, and the state, in the colour that outranks it. `be-best/assets/css/kanban.css` owns that grade, and this partial only states the facts it grades - the priority in the first character of the file name, the state in the dates the card carries. The banner is the one loud part of the face, and it is the state's: late is the danger colour, done is the success colour, and an open card on time is plain brand. The number in the circle is the same priority the frame states, so a card read close and a card read far agree. */}} {{- $page := . -}} {{- $id := replaceRE "^p[0-9]-" $page.File.BaseFileName "" -}} {{- $isCard := findRE `^p[0-9]-[a-z]{2}-[a-z0-9-]+$` $page.File.BaseFileName -}} {{- $priority := 9 -}} {{- with findRE "^p([0-9])-" $page.File.BaseFileName -}} {{- $priority = int (replaceRE "^p([0-9])-" "" (index . 0)) -}} {{- end -}} {{- $today := time.Format "2006-01-02" time.Now -}} {{- $completed := $page.Params.completionDate -}} {{- $overdue := false -}} {{- with $page.Params.due -}} {{- $overdue = and (not $completed) (gt $today (time.Format "2006-01-02" .)) -}} {{- end -}} {{- $state := "open" -}} {{- if $completed -}} {{- $state = "done" -}} {{- else if $overdue -}} {{- $state = "overdue" -}} {{- end -}} {{- $banner := "bg-brand" -}} {{- $bannerInk := "text-brand-ink" -}} {{- if $overdue -}} {{- $banner = "bg-danger" -}} {{- else if $completed -}} {{- $banner = "bg-success" -}} {{- end -}}
{{- if $isCard }} {{ $priority }} priority {{- end }}
