{{/* Render one level of a site menu as
  • items. dict: context - the current page menu - the menu entries to render at this level depth - 0-based depth of the caller maxDepth - deepest level rendered (default 3) ancestors - relPermalinks of the current page's ancestor sections */}} {{- $depth := add .depth 1 -}} {{- $maxDepth := default 3 .maxDepth -}} {{- $page := .context -}} {{- $ancestors := default (slice) .ancestors -}} {{- if le $depth $maxDepth -}} {{- range $element := .menu -}} {{- $entry := $element.Page -}} {{- $current := and $entry (eq $entry.RelPermalink $page.RelPermalink) -}} {{- $trail := and $entry (in $ancestors $entry.RelPermalink) -}} {{- $children := $element.Children -}} {{- $expanded := and $children (or $current $trail) -}} {{- if and $children (lt $depth $maxDepth) -}}
    • {{- partial "/navigation/recursive-nav.html" (dict "context" $page "menu" $children "depth" $depth "maxDepth" $maxDepth "ancestors" $ancestors ) -}}
    • {{- else -}} {{- $element.Params.icon | safeHTML -}} {{- $element.Name -}} {{- end -}} {{- end -}} {{- end -}}
