{{/* Render one level of a site menu as the
  • items of the top bar. dict: context - the current page menu - the menu entries to render at this level depth - 0-based depth of the caller maxDepth - deepest level rendered (default 3) ancestors - relPermalinks of the current page's ancestor sections An entry with children is a button that opens a panel; one without is a link. The entry of the current page is marked with aria-current="page", and the entries on the path to it are marked with is-trail, which is both a style and the state the drawer opens on. */}} {{- $depth := add .depth 1 -}} {{- $maxDepth := default 3 .maxDepth -}} {{- $page := .context -}} {{- $ancestors := default (slice) .ancestors -}} {{- if le $depth $maxDepth -}} {{- range $element := .menu -}} {{- $entry := $element.Page -}} {{- $current := and $entry (eq $entry.RelPermalink $page.RelPermalink) -}} {{- $trail := and $entry (in $ancestors $entry.RelPermalink) -}} {{- $children := $element.Children -}} {{/* A panel needs an id of its own for its trigger's aria-controls. The entry's link is unique, so it is a good basis; a URL entry adds its name in case two share a URL. */}} {{- $source := "" -}} {{- with $entry -}} {{- $source = .RelPermalink -}} {{- else -}} {{- $source = printf "%s %s" $element.Name $element.URL -}} {{- end -}} {{- $key := printf "%s" $source | anchorize -}} {{- $panelId := printf "nav-panel-d%d-%s" $depth $key -}} {{- if and $children (lt $depth $maxDepth) -}}
