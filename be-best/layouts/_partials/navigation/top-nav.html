{{/* The primary navigation: the site menus, as the horizontal list in the top bar and as the contents of the drawer below 1024 pixels. Which menus to show, how deep to nest them, and which menus sit behind one grouped entry, come from the `navigation` site params. Their entries - names, parents, weights and icons - come from the `menus` front matter of the content files. */}} {{- $nav := site.Params.navigation -}} {{- $menuNames := default (slice "main") $nav.menus -}} {{- $maxDepth := default 3 $nav.maxDepth -}} {{- $grouped := default (slice) $nav.grouped -}} {{- $groupedNames := slice -}} {{- range $group := $grouped -}} {{- $groupedNames = $groupedNames | append $group.menu -}} {{- end -}} {{- $currentPath := .RelPermalink -}} {{- $ancestors := slice -}} {{- range $ancestor := .Ancestors -}} {{- if and (ne $ancestor.RelPermalink "/") (ne $ancestor.RelPermalink $currentPath) -}} {{- $ancestors = $ancestors | append $ancestor.RelPermalink -}} {{- end -}} {{- end -}} {{/* The paths a menu entry has to be on to count as the current section: the current page and the sections above it. The root is not a section, so the home page leaves this list empty and no entry is on a trail. */}} {{- $trailPaths := slice -}} {{- range $ancestor := .Ancestors -}} {{- $trailPaths = $trailPaths | append $ancestor.RelPermalink -}} {{- end -}} {{- $trailPaths = $trailPaths | append $currentPath -}} {{/* A grouped menu sits behind one entry of its own, so the entry is on the trail when the page sits under any of that menu's links. */}}
    {{- range $menuName := $menuNames -}} {{- if not (in $groupedNames $menuName) -}} {{- with index site.Menus $menuName -}} {{- partial "/navigation/recursive-nav.html" (dict "context" $ "menu" . "depth" 0 "maxDepth" $maxDepth "ancestors" $ancestors ) -}} {{- end -}} {{- end -}} {{- end -}} {{- range $group := $grouped -}} {{- with index site.Menus $group.menu -}} {{- $trail := false -}} {{- range $entry := . -}} {{- with $entry.Page -}} {{- if in $trailPaths .RelPermalink -}} {{- $trail = true -}} {{- end -}} {{- end -}} {{- end -}} {{- $panelId := printf "nav-panel-menu-%s" ($group.menu | anchorize) -}}
